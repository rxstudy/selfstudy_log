\section{Definitions}

\textbf{Space of linear function} $L(V, W)$ vector space of linear functions from $V$ to $W$.

\textbf{Dual Space} $V^* = L(V, R)$. For each basis $\{e_i\}$ of $V$, there exists unique $\{e^i\}$ of $V^*$ such that $e^i (e_j) = \delta^i_j$

\textbf{Tensor Space} $T^r_s = \underbrace{V \otimes V}_\text{r times} \otimes \underbrace{V^* \otimes V^*}_\text{s times}  $  is space of multilinear functions on $$\underbrace{V^* \cross \ldots \cross V^*}_\text{r times} \cross \underbrace{V \cross \ldots \cross V}_\text{s times}$$

\textbf{Tensor Product} between $A$ of $(r,s)$ and $B$ of $(t, u)$,  is $$\begin{aligned}  A \otimes B &(\tau^1,\ldots, \tau^{r+t}, v_1,\ldots, v_{s+u}) \\ & = A(\tau^1, \ldots, \tau^r, v_1, \ldots, v_s) \\ &B(\tau^{r+1}, \ldots , \tau^{r+t}, v_{s+1}, \ldots, v_{s+u})\end{aligned}$$

\textbf{Vector Field} $X$ on coordinate neighborhood $U$ of a manifold $M$,  with coordinate $x^i$. For each point $p$, $X = X^i \partial_i$.  $X[f] = X^i \partial_i f$

\textbf{Change of Coordinates} If $Y$ has coordinate neighborhood $V$ of $y^i$, then $Y^i = X^j \frac{\partial y^i }{\partial x^j}$

\textbf{Map Differential(Pushforward)} $F_*$ is induced map $F_*: TM \rightarrow TN$ of $C^\infty$ map $F: M\rightarrow N$. $F_* (v_p) = (F_*v)_{F(p)}$. With coordinate, $F_* = [ \partial_j ( y^i \circ F ) ]$, the Jacobian of $F$. Note that $y^i \circ F = F^i(x^1, \ldots, x^m)$

\textbf{Tensor Bundle} $T^r_s M$  of type $(r,s)$ is the union of all tensor spaces $M^r_s(p)$ at each point $p \in M$.  

\textbf{Tangent Bundle} $TM=T^1_0M$, 

\textbf{Scalar Bundle} $T^0_0M = M \cross \real$, 

\textbf{Cotangent Bundle/ Differentials / Phase space} $T^0_1M$

\textbf{Tensor Field} $T$ of type $(r,s)$, $T(p) \in T^r_s M (p)$ for each $p$. $(1,0)$  is vector field, $(0,0)$ gives real-valued function. $(0, 1)$ gives differential.

\textbf{Tensor Coordinate} of $T^r_s$ wrt coordinate $x^i$ are $d^{r+s}$ real-valued functions $$T^{i_1\ldots i_r}_{j_1\ldots j_s} = T(dx^{i_1}, \ldots dx^{i_r}, \partial_{j_1}, \ldots , \partial_{j_s})$$

\textbf{Tensor Product}

\textbf{Alternating Tensor}: we can turn a $k$-tensor $T\in T^k(V^*)$ into an alternating tensor Alt$(T) \in \Lambda^k(V^*)$. by defining:
$$ \textnormal{Alt}(T) \equiv \frac{1}{k!} \sum_{\sigma \in S_k} (\textnormal{sgn} \sigma) (T\circ \sigma)$$
where $S_k = \{\sigma(v_{1}, \ldots, v_k) = (v_{\sigma(1)}, \ldots, v_{\sigma(k)})\}$ is the group of all permutation of the arguments. 


\textbf{Wedge Product}: After we construct the alternating tensors, we can define the wedge product between them.
$$ T\wedge S = \binom{k+m}{k} \textnormal{Alt}(T \otimes S)$$
where $T \in \Lambda^k(V^*)$, $S\in \Lambda^m(V^*)$ and $\textnormal{Alt}(T\wedge S) \in \Lambda^{k+m}(V^*)$.

\textbf{Pullback of a Covariant Tensor}: $$(f^* T)_p(v_1,\ldots, v_k) = T(f_* v_1, \ldots, f_* v_k) = T(df(v_1),\ldots, df(v_k))$$

\textbf{Lie Derivative of a tensor field}: Given a vector field $X$ on $M$, the Lie derivative of a $k$-covariant tensor field $T$ along $X$ is $$ L_XT \equiv \frac{d}{dt} (\psi_t^* T)\bigg{\rvert}_{t=0}$$, where $\psi_t = F(\cdot, t)$ with $F$ the local flow of $X$ at $p$.

\textit{REMARK:} The intuition is simple, imagine $T$ is a measurement device in a river placed around a neighborhood of point $p$ and the water is flowing along a vector field $X$, we see that $\psi_t^*T(v_1,\ldots, v_k) = T(\psi_{t*}v_1, \ldots, \psi_{t*} v_k)$ which pushes forward the measurement device by $t$ amount along the local flow around $p$. Then we take the limit as $t$ goes to 0 like we do with usual function derivative to get the rate of change of the measurement device's reading along the flow. 


\textbf{Exterior Product}

\textbf{Differential forms} p-form is $C^\infty$ skew-symmetric covariant tensor field of degree $p$ (type $(0,p)$). Local basis has ${d \choose p}$ p-forms $dx^{i_1}\cdots dx^{i_p}$ where $(i_1,\ldots,i_p)$ is increasing.

\subsection*{Musical Isomorphisms}
For a vector field $X = X^i\partial_i$ and a 1-form $\alpha = \alpha_i dx^i$:
\begin{align*}
	X^\flat &= g_{ij} X^j dx^i, \\
	\alpha^\sharp &= g^{ij} \alpha_j \partial_i.
\end{align*}

\subsection*{Hodge Star on $\mathbb{R}^3$ (Euclidean)}
For standard $dx, dy, dz$:
\begin{align*}
	\star 1 &= dx\wedge dy\wedge dz, \\
	\star dx &= dy\wedge dz, \quad \star dy = dz\wedge dx, \quad \star dz = dx\wedge dy, \\
	\star (dy\wedge dz) &= dx, \quad \star (dz\wedge dx) = dy, \quad \star (dx\wedge dy) = dz, \\
	\star (dx\wedge dy\wedge dz) &= 1.
\end{align*}

	\item $\operatorname{curl}(\operatorname{curl} X) = \operatorname{grad}(\operatorname{div} X) - \Delta X$ (Vector Laplacian). But I can skip this to keep it concise.

\subsubsection{Notations}
Projection operator: $\langle \rangle_p: \bigwedge(V) \to \bigwedge_p(V)$,  $$\langle A\rangle_p = A_{[p]} \in \bigwedge_p(V)$$ is the p-vector component of the multivector $A$

\begin{itemize}
	\item Grade Involution: $\sharp(A_{[p]}) = \widehat{A}_{[p]} = (-1)^p A_{[p]}$ \\
	\item Reversion: $\widetilde{(\bm{v}_1 \wedge \cdots \wedge \bm{v}_p)} = \bm{v}_p \wedge \cdots \wedge \bm{v}_1$ implies $\widetilde{A}_{[p]} = (-1)^{p(p-1)/2} A_{[p]}$
	\item Conjugation: $\bar{A}_{[p]} = \widehat{\widetilde{A}}_{[p]} = \widetilde{\widehat{A}}_{[p]}$
\end{itemize}



\subsection{Grassmann Algebra}
Flat operator $\flat: \bigwedge_p(V) \to \bigwedge^p(V)$ given by $$ (\bm{v}_1\wedge\cdots\wedge\bm{v}_p)_\flat = (\bm{v}_{1})_\flat \wedge \cdots \wedge (\bm{v}_{p})_\flat$$

Extension of the bilinear metric $G: \bigwedge_p(V)\times \bigwedge_p(V) \to \mathbb{R}$ given by 
$$
\begin{aligned}
	G(\bm{v}_1\wedge\cdots\wedge\bm{v}_p, \bm{u}_1\wedge\cdots\wedge\bm{u}_p) &= p!(\bm{v}_1\wedge\cdots\wedge\bm{v}_p)_\flat (\bm{u}_1\wedge\cdots\wedge\bm{u}_p) \\
	&= (\bm{v}_p\wedge\cdots\wedge\bm{v}_1)_\flat \rfloor (\bm{u}_1\wedge\cdots\wedge\bm{u}_p) \\
	&= \begin{Vmatrix}
		g(\bm{v}_1, \bm{u}_1) & \cdots & g(\bm{v}_1, \bm{u}_p) \\
		\vdots           &  \ddots  & \vdots \\
		g(\bm{v}_p, \bm{u}_1) & \cdots & g(\bm{v}_p, \bm{u}_p) \\
	\end{Vmatrix}
\end{aligned}
$$

$G(A_{[p]}, B_{[q]}) = 0$ when $p\neq q$

\section*{Hodge Star Identities}

\subsection*{Left Contraction}
$$\begin{aligned}
    (\alpha\rfloor A_{[p]})(\alpha_1, \ldots, \alpha_{p-1}) &= p A_{[p]}(\alpha,\alpha_1, \ldots, \alpha_{p-1}) \\
      &= p (\bm{v}_1 \wedge \ldots \wedge \bm{v}_p)(\alpha,\alpha_1, \ldots, \alpha_{p-1} ) \quad \quad \quad (\text{if } A_{[p]} = \bm{v}_1 \wedge \ldots \wedge \bm{v}_p ) \\
      &= p \frac{1}{p!} \sum_{\sigma \in S_p} \epsilon(\sigma) \alpha(\bm{v}_{\sigma(1)}) \cdots \alpha_{p-1}(\bm{v}_\sigma(p)) \\
      &=\frac{1}{(p-1)!} \begin{vmatrix}
      	\alpha(\bm{v_1}) & \ldots & \alpha(\bm{v_p}) \\
      	\alpha_1(\bm{v_1}) & \ldots & \alpha_1(\bm{v_p}) \\
      	\vdots & \ldots & \vdots \\
      	\alpha_{p-1}(\bm{v_1}) & \ldots & \alpha_{p-1}(\bm{v_p}) \\
      \end{vmatrix}
\end{aligned}$$

So $$
  (\alpha_1 \wedge \ldots \wedge\alpha_p)(\bm{v_1}\wedge \ldots \wedge \bm{v_p}) = \frac{1}{p!}  (\alpha_1 \wedge \ldots  \wedge \alpha_p)\rfloor (\bm{v_1}\wedge \ldots \wedge \bm{v_p}) = 
  \begin{vmatrix}
  \alpha_1(\bm{v_1}) & \ldots & \alpha_1(\bm{v_p}) \\
  \vdots & \ddots & \vdots \\
  \alpha_p(\bm{v_1}) & \ldots & \alpha_p(\bm{v_p}) \\
  \end{vmatrix}
$$

\subsubsection*{Properties}
\begin{itemize}
	\item $\alpha \rfloor (\bm{v} \wedge \bm{u}) = (\alpha \rfloor \bm{v}) \bm{u} - \bm{v} (\alpha \rfloor \bm{u}) = \alpha(\bm{v}) \bm{u} - \bm{v} \alpha(\bm{u}) $
	\item $\alpha \rfloor (A \wedge B) = (\alpha \rfloor A) \wedge B +  \hat{A} \wedge (\alpha \rfloor B)$ where $\hat{A}_{[p]} = \sharp A_{[p]} = (-1)^p A_{[p]}.$ (Graded Leibniz rule)
	\item $(\alpha_1 \wedge \ldots \wedge \alpha_p) \rfloor = \alpha_1 \rfloor  \alpha_2 \rfloor \ldots \alpha_p \rfloor$
\end{itemize}

\subsection*{Orientation}
For n linear independent vectors, $\bm{v_1}\wedge\ldots\wedge\bm{v_n} = J \bm{e_1}\wedge\ldots\wedge\bm{e_n}$. Orientation for $V^*$ and $V$ are $$\Omega_{V^*} = J' \bm{e^1}\wedge\ldots\wedge\bm{e^n}, \quad \quad \Omega_{V} = J \bm{e_1}\wedge\ldots\wedge\bm{e_n} $$
They form two equivalent classes: $$ \begin{aligned}
	[\Omega] &= \{A_{[n]} \in \bigwedge\nolimits_n(V) | A_{[n]} = J \Omega, \quad J > 0\} \\
	[-\Omega] &= \{A_{[n]} \in \bigwedge\nolimits_n(V) | A_{[n]} = J \Omega, \quad J < 0\} \\
\end{aligned}$$

\subsection*{Quasi-Hodge star}
Given $(V, \Omega_V)$, define 
$$
\begin{aligned}
	&\bar{\star}: \bigwedge\nolimits_p (V) \rightarrow \bigwedge\nolimits^{n-p}, \quad \quad &\Psi^{[p]} \wedge \bar{\star} A_{[p]} = p! \Psi^{[p]} (A_{[p]})\Omega_{V^*} \\
	&\underline{\star}: \bigwedge\nolimits^p (V) \rightarrow \bigwedge\nolimits_{n-p}, \quad \quad &A_{[p]} \wedge \underline{\star} \Psi^{[p]}  = p! \Psi^{[p]} (A_{[p]})\Omega_{V} \\
\end{aligned}
$$
Then we derive $$  \bar{\star} A_{[p]} = \tilde{A}_{[p]} \rfloor \Omega_{V^*}, \quad \quad \quad \underline{\star} \Psi^{[p]} = \tilde{\Psi}^{[p]} \rfloor \Omega_V$$



When $G$ is the Grassman extension of $g$ and $\Omega_V = \frac{1}{\sqrt{|\det g|}} e_1 \wedge \ldots \wedge e_n$(Pseudoscalar). Then we define hodge operator as composition of quasi hodge operator with the sharp/flat operator.
 $$\star = \sharp \circ \bar{\star} = \underline{\star} \circ \flat$$
It holds under the condition that $G(\Omega_V, \Omega_V) = 1$ which will make $\sharp \circ \bar{\star} = \underline{\star} \circ \flat$. 

Moreover $$ A\wedge\star B = G(A, B) \Omega_V, \quad \quad \star 1 = \Omega_V \quad \quad \star A = \tilde{A}_\flat \rfloor \Omega_V$$

For antiproduct(regressive product), we have $$ A_{[p]} \vee B_{[q]} = \star^{-1} (\star A_{[p]} \wedge \star B_{[q]})$$

The $\bigwedge^p(V)$ to $\bigwedge^{n-p}(V)$ version for forms is defined as map $\overset{*}{\star} = \flat \circ \star \circ \sharp$

\subsection*{Hodge Star Properties}
Let \((M^n, g)\) be an oriented Riemannian \(n\)-manifold with volume form \(\omega = \star 1\). For \(\alpha, \beta \in \Omega^k(M)\):

\begin{itemize}
	\item \textbf{Distributive (linear):} \(\star(\alpha + \beta) = \star\alpha + \star\beta\).
	\item \textbf{Double star:} \(\star\star\alpha = (-1)^{k(n-k)} \alpha\).
	\item \textbf{Not commutative:} In general, \(\star(\alpha \wedge \beta) \neq \star\alpha \wedge \star\beta\). Instead, the wedge product interacts via the metric:
	\[
	\alpha \wedge \star\beta = \beta \wedge \star\alpha = \langle \alpha, \beta \rangle \, \omega,
	\]
	where \(\langle \alpha, \beta \rangle = g^{i_1j_1}\cdots g^{i_kj_k} \alpha_{i_1\dots i_k}\beta_{j_1\dots j_k}\).
	\item \textbf{Volume identities:} \(\star 1 = \omega\), and \(\star \omega = 1\).
	\item \textbf{Codifferential:} \(\delta\alpha = (-1)^{n(k+1)+1} \star d \star \alpha\), satisfying \(\delta = -d\) on \(\Omega^0\) (up to sign conventions).
\end{itemize}


\subsection{Clifford Algebra}


$$
\begin{tikzcd}[scale=1.8, transform shape, column sep=large, row sep=large, cells={nodes={font=\Large}}]
	T(V) \arrow[rr, "\pi"']  & &  \mathrm{Cl}(V,g) \equiv \frac{T(V)}{I_{v\otimes v - g(v,v)\mathbf{1}}} \\
	& & \\
	\bigwedge(V) \equiv \frac{T(V)}{I_{v\otimes v}} \arrow[uu, "\iota"] \arrow[uurr, "\gamma= \pi \circ \iota"] &  &
\end{tikzcd}
$$

Here:
\begin{itemize}
	\item \(\pi: T(V) \to \mathrm{Cl}(V,g)\) is the Clifford projection
	\item \(\iota: \bigwedge(V) \to  T(V) \)  is the inclusion map,
	\item \(\gamma: \bigwedge(V) \rightarrow \mathrm{Cl}(V,g)\) is the bijective vector-space isomorphism \textbf{Clifford mapping}. It is natural to use the same notation used for exterior algebra for Clifford algebra. $\mathrm{Cl}(V, g) = \bigoplus^{n}_{p=0}\bigwedge_p(V)$ and projection operators is $\langle\rangle_p: \mathrm{CL}(V, g) \to \bigwedge_p (V)$ is equivalently defined.
\end{itemize}

Equivalently, \(\gamma\) is the Clifford algebra's "linear section" that identifies exterior wedge products with antisymmetrized Clifford products:
\[
\gamma(v_1 \wedge \cdots \wedge v_k) = \frac{1}{k!} \sum_{\sigma \in S_k} \mathrm{sgn}(\sigma) \, v_{\sigma(1)} \cdots v_{\sigma(k)}.
\]

\subsubsection{Properties}
\begin{itemize}
	\item $ \bm{vu} = v \wedge u + g(\bm{v}, \bm{u})$
	\item $\bm{vu} + \bm{uv} = 2g(\bm{v}, \bm{u})$
	\item $ \bm{v}A_{[p]} = \bm{v}\wedge A_{[p]} + \bm{v}_\flat \rfloor A_{[p]} \qquad \text{( or } A_{[p]}\bm{v} = A_{[p]} \wedge \bm{v} +  A_{[p]} \lfloor \bm{v}_\flat \text{ )} $
	\item $\bm{v} \wedge A = \frac{1}{2} (\bm{v}A + \hat{A}\bm{v})$
	\item $\bm{v}_\flat \rfloor A = \frac{1}{2}(\bm{v}A - \hat{A}\bm{v}) $
\end{itemize}

\subsection*{Cartan's Magic Formula (Lie derivative)}
For a vector field \(X\) and a differential form \(\alpha \in \Omega^k(M)\), the Lie derivative is:
\[
\mathcal{L}_X \alpha = d(\iota_X \alpha) + \iota_X(d\alpha),
\]
where \(\iota_X\) denotes interior multiplication. For a vector field \(Y\):
\[
\mathcal{L}_X Y = [X, Y].
\]

\subsection*{Cartan's Structure Equations}
Let \(\{e_i\}\) be a local orthonormal frame, \(\{\theta^i\}\) its dual coframe, and \(\omega^i_{\;j}\) the connection 1-forms defined by \(\nabla_X e_j = \omega^i_{\;j}(X) e_i\) (Levi-Civita connection, torsion-free). The curvature 2-forms are \(\Omega^i_{\;j}\).

\subsubsection*{Form versions}
\begin{align}
	&\text{First structure equation (torsion-free):} 
	d\theta^i = -\omega^i_{\;j} \wedge \theta^j. \label{eq:first_form} \\
	&\text{Second structure equation (curvature):} 
	\Omega^i_{\;j} = d\omega^i_{\;j} + \omega^i_{\;k} \wedge \omega^k_{\;j}. \label{eq:second_form}
\end{align}
The Riemannian curvature tensor satisfies \(\Omega^i_{\;j} = \frac{1}{2} R^i_{\;jkl} \theta^k \wedge \theta^l\), i.e. \(R(X,Y)Z = \iota_Z \Omega(X,Y)\) (index conventions may vary).

\subsubsection*{Vector field versions}
For vector fields \(X,Y,Z\):
\begin{align}
	&\text{Torsion tensor (zero for Levi-Civita):} 
	T(X,Y) = \nabla_X Y - \nabla_Y X - [X,Y]. \label{eq:torsion_vec} \\
	&\text{Curvature tensor (Riemann):} 
	R(X,Y)Z = \nabla_X\nabla_Y Z - \nabla_Y\nabla_X Z - \nabla_{[X,Y]}Z. \label{eq:curv_vec}
\end{align}
For a torsion-free connection, Equation~\eqref{eq:torsion_vec} simplifies to \(\nabla_X Y - \nabla_Y X = [X,Y]\), which is exactly the dual of Equation~\eqref{eq:first_form}.

\subsection*{Relation between form and vector versions}
The connection 1-form relates to the covariant derivative via:
\[
\omega^i_{\;j}(X) = \theta^i(\nabla_X e_j).
\]
The curvature forms relate to the Riemann tensor by:
\[
\Omega^i_{\;j}(X,Y) = \theta^i\bigl( R(X,Y) e_j \bigr).
\]
Equivalently, \(R(X,Y)Z = \iota_Z(\Omega^i_{\;j} \otimes e_i)\).

\subsection*{Definitions of the Differential Forms (Minkowski spacetime, metric \((-,+,+,+)\), SI units with \(c=1\))}

Let \(\mathbf{E} = E_x dx + E_y dy + E_z dz\) and \(\mathbf{B} = B_x dx + B_y dy + B_z dz\) be the electric and magnetic field 1-forms. Define the \textbf{Faraday 2-form} \(F\) and the \textbf{current 3-form} \(J\):
\[
F = \mathbf{E} \wedge dt + \star_3\mathbf{B} 
= E_x\, dx\wedge dt + E_y\, dy\wedge dt + E_z\, dz\wedge dt
+ B_x\, dy\wedge dz + B_y\, dz\wedge dx + B_z\, dx\wedge dy,
\]
\[
J = \rho\, \omega - \mathbf{J}\cdot d\mathbf{S}\wedge dt
= \rho\, dx\wedge dy\wedge dz - (J_x\, dy\wedge dz + J_y\, dz\wedge dx + J_z\, dx\wedge dy)\wedge dt,
\]
where \(\omega = dx\wedge dy\wedge dz\) is the Euclidean volume form, \(\star_3\) is the 3D Hodge star, and \(d\mathbf{S} = (dy\wedge dz,\; dz\wedge dx,\; dx\wedge dy)\).

The Hodge star \(\star\) in 4D gives the dual 2-form:
\[
\star F = -\mathbf{E}\cdot d\mathbf{S} + \mathbf{B}\wedge dt 
= -E_x\, dy\wedge dz - E_y\, dz\wedge dx - E_z\, dx\wedge dy
+ B_x\, dt\wedge dx + B_y\, dt\wedge dy + B_z\, dt\wedge dz.
\]

\subsection*{Compact Unified Form}

In differential forms, the four equations reduce to just two geometric statements:

\[
\boxed{dF = 0} \qquad \text{(Bianchi identity / homogeneous equations)},
\]
\[
\boxed{d\star F = \mu_0 J} \qquad \text{(source equation / inhomogeneous equations)}.
\]

\noindent
Expanding \(dF=0\):
\[
dF = (\nabla\cdot\mathbf{B})\, \omega \;+\; \left(\nabla\times\mathbf{E} + \frac{\partial\mathbf{B}}{\partial t}\right)\cdot d\mathbf{S}\wedge dt = 0,
\]
which directly yields both \(\nabla\cdot\mathbf{B}=0\) and \(\nabla\times\mathbf{E} = -\partial_t\mathbf{B}\).

Expanding \(d\star F = \mu_0 J\):
\[
d\star F = (\nabla\cdot\mathbf{E})\, \omega \;+\; \left(\nabla\times\mathbf{B} - \frac{\partial\mathbf{E}}{\partial t}\right)\cdot d\mathbf{S}\wedge dt 
= \mu_0 \rho\, \omega - \mu_0 \mathbf{J}\cdot d\mathbf{S}\wedge dt,
\]
which yields both \(\nabla\cdot\mathbf{E} = \rho/\epsilon_0\) (since \(\mu_0\epsilon_0=1\)) and \(\nabla\times\mathbf{B} = \mu_0\mathbf{J} + \partial_t\mathbf{E}\).

\subsection*{Reintroducing \(c\) and SI constants}

If one wants to restore the speed of light \(c\), define the Faraday 2-form with proper SI units:
\[
F = \mathbf{E}\wedge (c\,dt) + \star_3\mathbf{B} \quad\text{or}\quad 
F_{\text{SI}} = \frac{1}{c}\mathbf{E}\wedge dt + \star_3\mathbf{B}.
\]
With this normalisation, the equations become:
\[
dF = 0, \qquad d\star F = \mu_0 J,
\]
where \(J\) is defined exactly as above. Alternatively, in Gaussian units, the factor \(4\pi\) shifts between \(F\) and \(J\). The geometric structure \(dF=0,\; d\star F = J\) is universal; only the constants in the definition of \(J\) and \(F\) change.

\section*{Case Study 1: Surface of a sphere}
The surface of sphere of radius 1 is a manifold $$S^2 = \{(x, y, z) \in \Real^3 | x^2+y^2+z^2 = 1 \}$$
We can define a chart $(U, \psi)$ for $S^2$ where $U\subseteq M$ with spherical coordinate.  Let $$U = \left\{(\theta, \phi) \in [0, 2\pi]\times [0, \pi] \right\} $$
and $$
\psi(x,y,z): \begin{cases}
	\theta = \arccos(z) \\
	\phi = \text{sng}(y) \arccos \frac{x}{\sqrt{x^2 + y^2}}
\end{cases}, 
\psi^{-1}(\theta, \phi): 
\begin{cases}
	x &= \sin\theta \cos\phi \\
	y &= \sin\theta \sin\phi \\
	z &= \cos \theta
\end{cases} 
$$
Then $\psi(U) \subseteq \Real^2$ is a homeomorphism from $U$ to $\psi(U)$. $\psi$ is called a \textbf{Locale coordinate map}. And the component functions $(\theta, \phi)$ defined by $\psi(p) = (\theta(p), \phi(p))$ for $p \in S^2$ are called \textbf{local coordinates} on $U$.

One can think of this as giving a temporary identification between $U$ and $\psi(U)$. When we work in this chart, we can think of $U$ as an open subsets of the manifold and as an open subset of $\Real^2$. Thus, we can represent a point $p \in U \subseteq S^2$ by its coordinate $(\theta, \phi) = \psi(p)$ and think of it as being the point $p$. We say $(\theta, \phi)$ is the local coordinate for $p$ or $p = (\theta, \phi)$ in local coordinates. \textit{(See Lee's Smooth Manifold Local Coordinate Representations section)}

Given the same chart, the coordinate vectors $\partial_\theta, \partial_\phi$ form a basis for $T_pS^2$. If $v \in T_pS^2$, then  $$v = v^1 \frac{\partial}{\partial \theta}\bigg\rvert_p + v^2 \frac{\partial}{\partial \phi}\bigg\rvert_p = v^1\partial_\theta + v^2 \partial_\phi = v^i \partial_i $$

The dual space to $T_pS^2$ is $T^*_pS^2$, if $w \in T^*_pS^2$, $$w = w_1 d\theta + w_2 d\phi = w_i dx^i \text{(in generic coordinates)}$$
and $w(v) = w_i v^i$

$S^2$ is Riemannian with symmetric metric tensor defined as 
$$
\begin{aligned}
	g &= g_{ij}dx^i \otimes dx^j  \\
	  &= g_{11}d\theta\otimes d\phi + g_{12}d\theta \otimes d\phi + g_{21}d\phi \otimes d\theta + g_{22} d\phi \otimes d\phi \\
	  &= g_{11}(d\theta)^2 + \frac{1}{2}(g_{12} + g_{21}) d\theta \otimes d\phi +  \frac{1}{2}(g_{21} + g_{12})  d\phi\otimes d\theta +  g_{22} (d\phi)^2 \qquad ,(g_{12} = g_{22}) \\
	  &= g_{11}(d\theta)^2 + \frac{g_{12}}{2}(d\theta \otimes d\phi + d\phi\otimes d\theta) + \frac{g_{21}}{2}(d\phi\otimes d\theta + d\theta \otimes d\phi) + g_{22} (d\phi)^2 \\
	  &= g_{11}(d\theta)^2 + g_{12}d\theta d\phi + g_{21}d\phi d\theta + g_{22} (d\phi)^2 \\
	  &= g_{ij}dx^i dx^j
\end{aligned}$$

We will now compute $g$. Since $(\theta, \phi)$ are local coordinate of $S^2$, we can introduce a smooth immersion map $\iota = \psi^{-1} = (\sin\theta \cos\phi, \sin\theta\sin\phi, \cos\theta)$ into $\Real^3$. Since $\Real^3$ has Euclidean metric $\bar{g} = (dx)^2 + (dy)^2 + (dz)^2$, then $g$ is the pullback of $\bar{g}$,
$$\begin{aligned} 
	g &= \iota^*\bar{g} \\
	  &= (d(\sin\theta \cos\phi))^2 + (d(\sin\theta\sin\phi))^2 + (d(\cos\theta))^2\\
	  &= (\cos\theta \cos\phi d\theta - \sin\theta \sin\phi d\phi)^2 + (\cos\theta\sin\phi d\theta + \sin\theta\cos\phi d\phi)^2 + (\sin\theta d\theta))^2\\
	  &= (d\theta)^2 + \sin^2\theta (d\phi)^2
\end{aligned} 
$$


\section*{Case Study 2: Relativistic length contraction}
Given a stationary frame, it has Minkowski flat metric of $(d\tau)^2 = (dt)^2 - (dx)^2$.
A measure of length between location $A$ and $B$ along $x$-axis in a stationary frame are the distance between two simultaneous events $E_A = (t_0, x_a)^T$ and $E_B = (t_0, x_b)^T$. We want to calculate the distance with respect to a moving frame with constant velocity.

Let $S$ be the stationary frame with axis $(t, x)$ and $S'$ with axis $(\bar{t}, \bar{x})$ be the moving frame in the $x$-direction with speed $v$. The line element $d\tau$ is invariant in different frames, therefore $(dt)^2 - (dx)^2 = (d\tau)^2 = (d\bar{t})^2 - (d\bar{x})^2$. So we have $(t)^2 - (x)^2 = (\bar{t})^2 - (\bar{x})^2$. The solution is given by $$
 \begin{aligned}
 	t &= \bar{t}\cosh \theta + \bar{x} \sinh\theta   & \\
 	x &= \bar{t}\sinh \theta + \bar{x} \cosh\theta   & 
 \end{aligned}
$$
The trajectory of the origin of $S'$ along x axis is $x(t) = vt$ in frame $S$ but $\bar{x}(\bar{t}) = 0$ in $S'$ after the above transformation. So the above transformation is mapping $(\bar{t}, 0)$ to $(t, vt)$. Substituting those in the solution, then we have $v = \tanh\theta \equiv \beta$. From that, we have $\cosh \theta = \frac{1}{\sqrt{1 - v^2}} \equiv \gamma$, $\sinh\theta = \frac{v}{\sqrt{1 - v^2}} = \gamma\beta$ and . We arrive at Lorentz transform from $S \rightarrow S'$: $$\Lambda = \begin{pmatrix}
	\cosh\theta & \sinh\theta \\
	\sinh\theta & \cosh\theta \\ 
\end{pmatrix}^{-1}
= \begin{pmatrix}
	\cosh\theta & - \sinh\theta \\
	-\sinh\theta & \cosh\theta \\ 
\end{pmatrix}
= \begin{pmatrix}
	\gamma & -\gamma\beta \\
	-\gamma \beta & \gamma \\
\end{pmatrix}
$$

Now consider the world line of $A$ and $B$ in $S$,  
$$W_A(t) = \begin{pmatrix}
	t \\ x_a
\end{pmatrix},  
W_B(t) =  \begin{pmatrix}
t \\  x_b
\end{pmatrix}$$. After Lorentz transform, $$
\widehat{W}_A(t) = \begin{pmatrix}
 \gamma t - \gamma \beta x_a \\
 - \gamma\beta t + \gamma x_a
\end{pmatrix},
\widehat{W}_B(t) = \begin{pmatrix}
	\gamma t - \gamma \beta x_b \\
	- \gamma\beta t + \gamma x_b
\end{pmatrix}
$$
As we can see when $E_A$ and $E_B$ that are simultaneous in $S$ is not simultaneous in $S'$ because their time component is not the same. To measure the distance in $S'$, we will choose two events along the world line of $A$ and $B$ with the same time component in $S'$ that is $\widehat{W}_A^{(0)}(t_a) = \widehat{W}_B^{(0)}(t_b)$. Therefore we have $\gamma t_a  -  \gamma\beta x_a = \gamma t_b - \gamma \beta x_b$. We can choose $t_a = \beta x_a$ and $t_b = \beta x_b$ which is their  $\bar{x}$-intercept in $S'$.

Then the distance measured in $S'$ is $$\begin{aligned}
\widehat{L} & =  \widehat{W}_B^{(1)}(\beta x_b) - \widehat{W}_A^{(1)}(\beta x_a)\\
   & = -\gamma\beta^2 x_b + \gamma x_b + \gamma \beta^2 x_a - \gamma x_a  = \gamma(1 - \beta^2)(x_b - x_a) \\
   &= \frac{x_b - x_a}{\gamma}\\
   &= \frac{L}{\gamma}
\end{aligned}$$
