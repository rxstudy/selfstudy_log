\section{Basic Elementary Principles}

\begin{theorem}(Newton's law)
	\begin{enumerate}
		\item  \textnormal{In an inertial reference frame, an object remains at rest or constant velocity unless acted upon by external force.}
		\item  $$\dot{\bb{p}} = \frac{d(m\bb{v})}{dt} = \bb{F}^{(e)}$$
		\item  \textnormal{Action and reaction: $\bb{F}_{ij} = -\bb{F}_{ji}$ (Additional condition for strong version: $\bb{r}_{ij} \times \bb{F}_{ji} = 0$)}
	\end{enumerate}
\end{theorem}
\subsection{Single particle}

 \begin{theorem}[Conservation of Linear Momentum]
 	$$\text{If } \bb{F} = 0, \text{ then } \bb{\dot p} = 0$$
 \end{theorem}

Angular momentum of the particle around point $O$ is $\bb{L} = \bb{r} \times \bb{p}$. Torque $\bb{N} = \bb{r} \times \bb{F}$. 

$$ \bb{N} = \frac{d}{dt} (\bb{r} \times m\bb{v}) = \frac{d \bb{L}}{dt} \equiv \bb{\dot L}$$

\begin{theorem}[Conservation of Angular Momentum]
	$$\textnormal{If $\bb{N} = 0$,  then $\bb{\dot L} = 0$} $$
\end{theorem}

If force field $\bb{F}$ is conservative ($\oint \bb{F} \cdot d \bb{s} = 0$ or $\int_a^b \bb{F} \cdot d\bb{s} = T_b - T_a$ is the same for any path between $a$ and $b$). 
There exists a potential scalar field $V$ such that $$\bb{F} = -\grad{V(\bb{r})}$$

\begin{theorem}[Conservation of Energy]
	 $$E = T + V \textnormal{ is conserved where } T \textnormal{ and } V \textnormal{ are kinetic energy and potential energy respectively}$$
\end{theorem}

\subsection{System of particles}
Center of mass: $$\bb{R} = \frac{\sum_i m_i \bb{r}_i}{\sum m_i} =  \frac{\sum_i m_i \bb{r}_i}{M}$$

Let $\bb{F}^{(e)}_i$ external force acting on particle $i$-th, and $\bb{F}_{ji}$ is the force 
exerted by $j$-th particle on $i$-th particle in the system, 
$$  m_i \frac{d^2(\bb{r_i})}{dt^2} = \bb{\dot{p_i}} = \sum_{j} \bb{F}_{ji} +  \bb{F}^{(e)}_i  $$
Summing over all particles, we get $$ \bb{\dot{p}} =  M \frac{d^2 \bb{R}}{dt^2} = \sum_i \bb{F}^{(e)}_i \equiv \bb{F}^{(e)}$$

\begin{theorem}[Conservation of Linear Momentum for a system of particles]
	$$\textnormal{if the total external force $\bb{F}^{(e)} = 0$, the total linear momentum $\bb{p}$ is conserved.}$$
\end{theorem}

The change of total angular momentum: $$\begin{aligned}
 \frac{d}{dt}\bb{L} &= \sum_i \bb{r_i}\times \bb{\dot{p}_i} = \sum_i \bb{r_i} \times (\bb{F}^{(e)}_i + \sum_j \bb{F}_{ji}) \\
  &= \bb{N}^{(e)}   + \sum_{i, j} \bb{r_i} \times \bb{F}_{ji} \\
  &= \bb{N}^{(e)}   + \sum_{i < j} \bb{r_i} \times \bb{F}_{ji} + \sum_{j < i} \bb{r_i} \times \bb{F}_{ji} \\
  &= \bb{N}^{(e)}   + \sum_{i < j} \bb{r_i} \times \bb{F}_{ji} + \sum_{i < j} \bb{r_j} \times \bb{F}_{ij} \\
  &= \bb{N}^{(e)}   + \sum_{i < j} \bb{r_i} \times \bb{F}_{ji} - \sum_{i < j} \bb{r_j} \times \bb{F}_{ji} \\
  &= \bb{N}^{(e)}  + \sum_{i < j} \left(\bb{r_i} - \bb{r_j} \right) \times \bb{F}_{ji} \\
  &= \bb{N}^{(e)}  + \sum_{i < j} \bb{r}_{ij} \times \bb{F}_{ji} \\
\end{aligned}$$

When the internal force abides strong version of Newton's 3rd law, internal force pair has the same direction as the two particles $\bb{r}_{ij}$. Then $\bb{r}_{ij}  \times \bb{F}_{ji} = 0$.  Forces satisfy strong Newton's 3rd law are \textbf{central}.
\begin{theorem}[Conservation of Angular Momentum for a system of particles]
   $$\textnormal{$\bb{L}$ is constant in time if applied external torque $\bb{N}^{(e)} = 0$ and internal forces are central}$$
\end{theorem}

Let $\bb{r}_i = \bb{R} + \bb{r}_i'$ where $\bb{r}_i'$ is the position of ith particle wrt to $\bb{R}$ instead of $O$. Similarly, $\bb{v}_i = \bb{v} + \bb{v}_i'$ where $\bb{v} = \dot{\bb{R}}$.  We can find $\bb{R} = \frac{\sum_i m_i \bb{r}_i}{M} = \frac{\sum_i m_i (\bb{R} + \bb{r}_i')}{M} = \bb{R} + \sum_i \bb{r}_i' $ which means $$\sum_i  \bb{r}_i' = 0, \quad\quad \sum_i  \bb{\dot{r}}_i' = \sum_i  \bb{v}_i' = 0 $$.

\begin{theorem}
\textnormal{The total angular momentum about $O$ is the angular momentum of center of mass + angular momentum of each particle around the center of mass.}
$$\begin{aligned}
	\bb{L} &=  \sum_i \bb{r_i}\times \bb{p_i} \\
	       &=\bb{R}\times M \bb{v} + \sum_i \bb{r}_i'\times \bb{p}_i' + \sum_i \bb{R}\times m_i \bb{v}_i' + \sum_i m_i \bb{r}_i' \times \bb{v} \\
	       &=    \bb{R}\times M \bb{v} + \sum_i \bb{r}_i'\times \bb{p}_i'
\end{aligned}
$$
\end{theorem}

In general $\bb{L}$ depends on choice of $O$ through $\bb{R}$ except when the center of mass is at rest $\bb{R} = 0$.

\begin{theorem}
\textnormal{The total kinetic energy of the system is }$$
T = \sum_i \frac{1}{2}m_i v_i^2 =  \sum_i \frac{1}{2} m_i (\bb{v} + \bb{v}_i') (\bb{v} + \bb{v}_i') = \frac{1}{2} M v^2 + \sum_i \frac{1}{2}m_i (v_i')^2 
$$
\end{theorem}

The ith particle has trajectory $\bb{r}_i(t)$ from time $a$ to $b$. Then the work is the change in kinetic energy.
$$W_i = \int_a^b \bb{F}_i  \cdot  d\bb{r_i} =  \int_a^b \frac{d(m_i v_i)}{dt}   \cdot  \frac{d \bb{r_i}}{dt} dt =  \int_a^b  m_i \bb{\dot{v}}_i  \cdot \bb{v}_i dt =  \int_a^b  d(\frac{1}{2} m_i v_i^2) = T_i(b) - T_i(a)$$

Also we can write 
$$
 W_i = \int_a^b \bb{F}_i  \cdot  d\bb{r_i}  = \int_a^b \bb{F}_i^{(e)}  \cdot  d\bb{r_i}  +  \sum_j \int_a^b \bb{F}_{ji} \cdot  d\bb{r_i}
$$

If the external force $\bb{F}_i^{(e)}$ is conservative, then $\bb{F}_i^{(e)} = -\grad{V_i}$

$$
   \int_a^b \bb{F}_i^{(e)}  \cdot  d\bb{r_i}
    = - \int_a^b \grad{V_i} \cdot  d\bb{r_i} 
    = - \int_a^b \frac{\partial V_i}{\partial x_i} dx_i + \frac{\partial V_i}{\partial y_i} dy_i +  \frac{\partial V_i}{\partial z_i} dz_i 
    = -\int_a^b dV_i 
    =- V_i |^b_a 
$$

If the action/reaction forces are conservative and central,  $V_{ji}(|\bb{r_i} -\bb{r_j}|)$ only depends on distance between ith and jth particle, hence it is symmetric $V_{ji} = V_{ij}$. Also 
$$\bb{F}_{ji} = - \grad_{i}{V_{ij}} = - \bb{F}_{ij} =  \grad_{j}{V_{ji}} = \grad_{j}{V_{ij}} $$ 


We can find the gradients of $V_{ij}$ with respect to $\bb{r_{ij}} =\bb{r_i} -\bb{r_j}$, 

$$
 \grad_{i}{V_{ij}} = \frac{\partial V_{ij}}{\partial \bb{r_i}} = \frac{\partial V_{ij}}{\partial \bb{r_{ij}}} \frac{\partial   \bb{r_{ij}}}{\partial \bb{r_i}} = \frac{\partial V_{ij}}{\partial \bb{r_{ij}}} = \grad_{ij}{V_{ij}} = - \grad_{j}{V_{ij}}
$$

 The work done by internal forces can be written in sums of pairs of potentials between ith and jth particle.
$$
\begin{aligned}
	\sum_{i, j} \int_a^b \bb{F}_{ji} \cdot  d\bb{r_i}
	&= -\sum_{i, j} \int_a^b  \grad_{j}{V_{ji}} \cdot  d\bb{r_i} \\
	&= -\sum_{i < j} \int_a^b   \grad_{j}{V_{ji}} \cdot  d\bb{r_i} +  \grad_{i}{V_{ij}} \cdot  d\bb{r_j} \\
	&= -\sum_{i < j} \int_a^b   \grad_{j}{V_{ij}} \cdot  d\bb{r_i} -  \grad_{j}{V_{ij}} \cdot  d\bb{r_j} \\
	&= -\sum_{i < j} \int_a^b   \grad_{j}{V_{ij}} d(\bb{r_i} - \bb{r_j}) \\
	&= -\sum_{i < j} \int_a^b   \grad_{j}{V_{ij}} d(\bb{r_{ij}}) \\
	&=  \sum_{i < j} \int_a^b   \grad_{ij}{V_{ij}} d(\bb{r_{ij}}) \quad \quad (\textnormal{note that }\grad_{ij}{V_{ij}} = - \grad_{j}{V_{ij}}) \\
	&=  \frac{1}{2} \sum_{i,j} \int_a^b   \grad_{ij}{V_{ij}} d(\bb{r_{ij}}) \\
	&=  \frac{1}{2} \sum_{i,j} {V_{ij}} |^b_a
\end{aligned}
$$

\begin{theorem}
	\textnormal{Total potential energy $V$ when external and internal forces are conservative and internal forces also central is:}
	$$ V = \sum_i V_i  + \frac{1}{2} \sum_{i,j} {V_{ij}}$$
	
	\textnormal{Such that total energy $T + V$ is conserved}
\end{theorem}

On a rigid body, $|\bb{r_{ij}}|$ is constant between any two points. Then $0 = d(r_{ij}^2) = d(\bb{r}_{ij} \cdot \bb{r}_{ij}) = 2 \bb{r}_{ij} \cdot d(\bb{r}_{ij})$. Therefore $d(\bb{r}_{ij})$ is perpendicular to $\bb{r}_{ij}$ which means the work done by internal force $\bb{F}_{ji} \cdot d(\bb{r}_{ij}) = 0$

\subsection{Constraints}
\textbf{Holonomic Constraints} have the form $$f(\bb{r_1}, \bb{r_2}, \bb{r_3}, \ldots, t) = 0$$
For rigid body, $r_{ij}^2 - c_{ij}^2 = 0$. Constraints can't be expressed as above are called non-holonomic (e.g. placing a particle on a sphere surface is non-holonomic: $r^2 - a^2 \geq 0$, it will slide and eventually fall off the sphere).

\textbf{Rheonomous Constraints} contains time as an explicit variable. \textbf{Scleronomous Constraints} are not explicitly dependent on time.

Constraints difficulties:
\begin{enumerate}
	\item the coordinates $r_i$ are no longer independent.
	\item forces of constraint are unknown a priori but we know the effect. e.g. force of wall on gas particle.
\end{enumerate}

\textbf{Generalized Coordinates} can solve problem 1. Consider a system of $N$ particles without constraints, it has $3N$ independent coordinate (\textbf{degrees of freedom}). When there are $K$ \textbf{holonomic} constraints, we only have $3N-K$ degree of freedom, so we can re-parameterize the system with $q_1,\ldots,q_{3N-K}$ independent variables. 
$$  \bb{r}_i = \bb{r}_i(q_1, q_2, \ldots, q_{3N-k}, t), \quad \textnormal{ for $1\leq i \leq N$}$$
If the constraints are non-holonomic, we cannot eliminate the dependent variables this way. 

\subsection{D'Alembert Principle and Lagrangian equations}

\textbf{virtual displacement} of a system is a change of the configuration of the system as the result of any arbitrary infinitesimal change of the coordinates:  $\delta \bb{r}_i$, \textit{consistent with the forces and constraints imposed on the system at the given instant $t$}.  

This is different from actual displacement of a system in time interval $dt$ during which the force and constraints might be changing.

\textbf{When system is in equilibrium}, total force on each partcile $\bb{F}_i = 0$. So the \textbf{virtual work} $\bb{F}_i \cdot \delta \bb{r}_i = 0$.  Note that total force is the sum of applied force and constraint force: $\bb{F_i} = \bb{F}^{(a)}_i +  \bb{f}_i$. Therefore
$$ \sum_i \bb{F}_i \cdot \delta \bb{r}_i =   \sum_i \bb{F}^{(a)}_i \cdot \delta \bb{r}_i +  \sum_i \bb{f}_i \cdot \delta \bb{r}_i = 0$$

We are actually picking the generalized coordinates $q_i$ to get rid of the constraint equations so they become trivial $0=0$ on both side

\begin{theorem}[Principle of virtual work]
\textnormal{We assume \textbf{the total net virtual work done by constraint forces are zero} which is $\sum_i \bb{f}_i \cdot \delta \bb{r}_i = 0$,  we get the principle of virtual work for equilibrium for a system: $$ \sum_i \bb{F}^{(a)}_i \cdot \delta \bb{r}_i  = 0$$}
\end{theorem}
In general, $\bb{F}^{(a)}_i \neq 0$ and $\delta \bb{r}_i$ are not independent, so they can't vary freely. Not much can be said about them.

Note that $\bb{F}_i = \dot{\bb{p}}_i \Rightarrow \bb{F}_i - \dot{\bb{p}}_i = 0$, we can write 
$$
 \begin{aligned}
   &\quad \sum_i (\bb{F}_i - \dot{\bb{p}}_i) \cdot \delta \bb{r}_i  = 0 \\
   &\Rightarrow \sum_i ( \bb{F}^{(a)}_i  - \bb{\dot{p}}_i  + \bb{f}_i ) \cdot \delta \bb{r}_i = 0 \\
   &\Rightarrow \sum_i ( \bb{F}^{(a)}_i  - \bb{\dot{p}}_i) \cdot \delta \bb{r}_i + \bb{f}_i \cdot \delta \bb{r}_i = 0 \\ 
   &\Rightarrow \sum_i ( \bb{F}^{(a)}_i  - \bb{\dot{p}}_i) \cdot \delta \bb{r}_i = 0 \\
 \end{aligned}
$$

\begin{theorem}[D'Alembert Principle]
\textnormal{ For an equilibrium system where the total net virtual work done by constraint forces are zero, we have $$\sum_i ( \bb{F}^{(a)}_i  - \bb{\dot{p}}_i) \cdot \delta \bb{r}_i = 0$$}
\end{theorem}

Suppose $\bb{r}_i = \bb{r}_i(q_1, \ldots, q_n, t)$,  we have velocity, $$\bb{v}_i = \bb{\dot{r}}_i = \sum_j \frac{\partial \bb{r}_i}{\partial q_j} \dot{q}_j + \frac{\partial \bb{r}_i}{\partial t} $$
and $$ \delta \bb{r}_i = \sum_j \frac{\partial \bb{r_i}}{\partial q_j} \delta q_j$$

Note that 
$$\frac{\partial \bb{v}_i}{\partial \dot{q}_k} = \frac{\partial \bb{r}_i}{\partial q_j} \delta^k_j =  \frac{\partial \bb{r}_i}{\partial q_k}, \quad \quad \frac{d}{dt}\left(\frac{\partial \bb{r}_i}{\partial q_j} \right) 
= \sum_k \frac{\partial^2 \bb{r}_i}{\partial q_j \partial q_k} \dot{q}_k + \frac{\partial^2 \bb{r}_i}{\partial q_j \partial t} 
= \frac{\partial \bb{v}_i}{\partial q_j} 
= \frac{\partial \bb{\dot{r}}_i}{\partial q_j} $$

Then we have the following:
$$
  \begin{aligned}
  	&\quad \sum_i \bb{\dot{p}}_i \cdot \delta \bb{r}_i  \\
  	&=  \sum_i (m_i \dot{\bb{v}}_i) \cdot  \sum_j \frac{\partial \bb{r_i}}{\partial q_j} \delta q_j  \\
  	&=  \sum_j \sum_i m_i \frac{d\bb{v}_i}{dt}  \cdot  \frac{\partial \bb{v_i}}{\partial \dot{q}_j} \delta q_j  \\
  	&=  \sum_j \left\{  \sum_i \left( m_i \frac{d \bb{v}_i}{dt}  \cdot  \frac{\partial \bb{v_i}}{\partial \dot{q}_j}  +    m_i \bb{v}_i  \cdot \frac{d}{dt} \frac{\partial \bb{v_i}}{\partial \dot{q}_j}\right) - m_i \bb{v}_i  \cdot \frac{d}{dt} \frac{\partial \bb{v_i}}{\partial \dot{q}_j} \right\} \delta q_j, 
  	\quad \textnormal{(adding 0)}\\
  	&= \sum_j \left\{     
  	\sum_i 
  	\frac{d}{dt}\left(m_i \bb{v}_i  \cdot   \frac{\partial \bb{v}_i}{\partial \dot{q}_j} \right) - m_i \bb{v}_i  \cdot   \frac{\partial \bb{v}_i}{\partial q_j}
  	 \right\}  \delta q_j,
  	  \quad \textnormal{(chain rule)}  \\
  	&= \sum_j \left\{     
  	\sum_i 
  	\frac{d}{dt}\left( \frac{1}{2} m_i  \frac{\partial (\bb{v}_i  \cdot \bb{v}_i)}{\partial \dot{q}_j} \right) - \frac{1}{2} m_i \frac{\partial (\bb{v}_i  \cdot \bb{v}_i)}{\partial q_j}
    \right\}\delta q_j  \\
  	&= \sum_j \left\{     
  	\sum_i
  	\frac{d}{dt}\left(  \frac{\partial T_i }{\partial \dot{q}_j} \right) -  \frac{\partial T_i}{\partial q_j}
  	 \right\} \delta q_j, 
  	 \quad \textnormal{($T_i$ is the kinetic energy $\frac{1}{2}m_iv_i^2$)} \\
  	&= \sum_j \left\{ 
  	\frac{d}{dt}\left(  \frac{\partial T}{\partial \dot{q}_j} \right) -  \frac{\partial T}{\partial q_j}
  	 \right\}  \delta q_j\\
  \end{aligned}
$$


We define \textbf{Generalized force} $Q_j$ such that 
$$ \sum_i \bb{F}_i \cdot \delta \bb{r}_i 
=  \sum_j   \left( \sum_i \bb{F}_i \cdot  \frac{\partial \bb{r}_i}{\partial q_j} \right) \delta q_j 
 \equiv \sum_j   Q_j \delta q_j $$

\begin{theorem}
\textnormal{Then the D'Alember principle equation becomes}
$$
  \begin{aligned}
  	&\quad \sum_i ( \bb{F}^{(a)}_i  - \bb{\dot{p}}_i) \cdot \delta \bb{r}_i = 0 \\
  	&\Rightarrow \sum_j \left\{ 
  	\frac{d}{dt}\left(  \frac{\partial T}{\partial \dot{q}_j} \right) -  \frac{\partial T}{\partial q_j} - Q_j
  	 \right\}\delta q_j = 0 \\
  	&\Rightarrow 	\frac{d}{dt}\left(  \frac{\partial T}{\partial \dot{q}_j} \right) -  \frac{\partial T}{\partial q_j} - Q_j = 0 \\
  	&\Rightarrow 	\frac{d}{dt}\left(  \frac{\partial T}{\partial \dot{q}_j} \right) -  \frac{\partial T}{\partial q_j} = Q_j 
  \end{aligned}
$$
Note: because each $\delta q_j$ is independent of one another, so the coefficient must vanish.
\end{theorem}

The term  $\frac{\partial T}{\partial q_j}$  is due to the curvature of the generalized coordinate $q_j$ (would be 0 in Cartesian coordinate).

Now suppose the applied forces are conservatives which means $\bb{F}_i = -\grad V_i$,  then $$
Q_j = \sum_i \bb{F}_i \cdot \frac{\partial \bb{r}_i}{\partial q_j} 
= - \sum_i  \grad V_i \cdot \frac{\partial \bb{r}_i}{\partial q_j} 
= - \sum_{i, k}  \frac{\partial V_i}{\partial r_i^k} \frac{\partial r_i^k}{\partial q_j} 
\equiv - \frac{\partial V}{\partial q_j}$$
Where $V(\bb{r}_1, \bb{r}_2, \ldots, \bb{r}_N, t)$. 

Then $$
\begin{aligned}
	&\quad \quad \frac{d}{dt}\left(  \frac{\partial T}{\partial \dot{q}_j} \right) -  \frac{\partial T}{\partial q_j} -  Q_j = 0 \\ 
	&\Rightarrow   \frac{d}{dt}\left(  \frac{\partial T}{\partial \dot{q}_j} \right) -  \frac{\partial T}{\partial q_j}  + \frac{\partial V}{\partial q_j} = 0 \\
	&\Rightarrow   \frac{d}{dt}\left(  \frac{\partial T}{\partial \dot{q}_j} \right) -  \frac{\partial (T - V)}{\partial q_j} = 0 \\
	&\Rightarrow   \frac{d}{dt}\left(  \frac{\partial (T - V)}{\partial \dot{q}_j} \right) -  \frac{\partial (T - V)}{\partial q_j} = 0
	,\quad \textnormal{(Because $\frac{\partial V}{\partial \dot{q}_j} = 0$ )}
\end{aligned}
$$

\begin{theorem}[Lagrange's equation]
\textnormal{If the constraints forces do zero net virtual work and the applied forces are conservative. We have:
  $$ \frac{d}{dt} \left(\frac{\partial L}{\partial \dot{q}_j}\right) - \frac{\partial L}{\partial q_j} = 0$$
Where the lagrangian $L = T - V$. 
}
\end{theorem}

\subsection{Velocity-dependent potentials and disspipation function}

\begin{theorem}[Velocity dependent potential]
\textnormal{
	The lagrangian equation
	$$ \frac{d}{dt} \left(\frac{\partial L}{\partial \dot{q}_j}\right) - \frac{\partial L}{\partial q_j} = 0$$
 is valid even if the generalized forces are obtained from $U(q_j, \dot{q_j})$ where $$Q_j = -\frac{\partial U}{\partial q_j} + \frac{d}{dt}\left(\frac{\partial U}{\partial \dot{q}_j}\right)$$ $L$ will be $T-U$. $U$ is called a generalized potential.
}
\end{theorem}

\begin{theorem}[Non potential force]
\textnormal{
If not all forces arise from a potential, then the following form of lagrangian is still valid. 
$$ \frac{d}{dt} \left(\frac{\partial L}{\partial \dot{q}_j}\right) - \frac{\partial L}{\partial q_j} = Q_i$$
Where $L$ contains the potential of conservative forces as before but $Q_i$ represent forces not arising from a  potential (i.e Frictions).
}
\end{theorem}

Frictional forces are usually proportional to velocity of the particle (i.e $f_x = -k_x v_x$). Such type of friction can be derived from $$\mathcal{F} = \frac{1}{2}\sum_i^{\textnormal{\# of particles}} (k_x v_{ix}^2 + k_y v_{iy}^2 + k_z v_{iz}^2)$$ Such $\mathcal{F}$ is called Rayleigh's dissipation function. Then the friction force is $$ \bb{F}_f = -\grad_v \mathcal{F} $$ 
For example, $\frac{\partial \mathcal{F}}{\partial v_{1x}} = - k_x v_{1x}$ is the x-direction friction force on the first particle.

The work done by the friction force is $2\mathcal{F}$:
$$ 
\begin{aligned}
 dW_f = \bb{F}_f \cdot d\bb{r} &= \bb{F}_f \cdot \frac{\partial \bb{r}}{\partial t} dt \\
  &=  - \bb{F}_f \cdot \bb{v} dt \\ 
  &= \grad_v \mathcal{F} \cdot \bb{v} dt \\ 
  &= \sum_i \sum_{j \in \{x,y,z\}} \frac{\partial \mathcal{F}}{\partial v_{ij}}v_j   \\
  &= \sum_i \sum_{j \in \{x,y,z\}} k_j (v_{ij})^2  dt \\
  &= 2 \mathcal{F} dt
\end{aligned}$$

To find the generalized force $Q_i$ for such friction: Note that $\frac{\partial \bb{r_i}}{\partial q_j} = \frac{\partial \bb{v_i}}{\partial \dot{q}_j}$ from previous derivations.
$$
\begin{aligned}
	Q_j &= \sum_i (\bb{F_f})_i \cdot \frac{\partial \bb{r}_i }{\partial q_j} = - \sum_i (\grad_v \mathcal{F})_i \cdot \frac{\partial \bb{r}_i }{\partial q_j}  \\ 
	&=-\sum_i \left( \frac{\partial \mathcal{F}}{\partial v_{ix}} \frac{\partial r_{ix} }{\partial q_j} + \frac{\partial \mathcal{F}}{\partial v_{iy}} \frac{\partial r_{iy} }{\partial q_j} + \frac{\partial \mathcal{F}}{\partial v_{iz}} \frac{\partial r_{iz} }{\partial q_j}\right) \\
	&=-\sum_i \left( \frac{\partial \mathcal{F}}{\partial v_{ix}} \frac{\partial v_{ix} }{\partial \dot{q}_j} + \frac{\partial \mathcal{F}}{\partial v_{iy}} \frac{\partial v_{iy} }{\partial \dot{q}_j} + \frac{\partial \mathcal{F}}{\partial v_{iz}} \frac{\partial v_{iz} }{\partial \dot{q}_j}\right) \\ 
	&= -  \frac{\partial \mathcal{F}}{\partial  \dot{q}_j}
\end{aligned}
$$

\begin{theorem}[Lagrangian with dissipation function]
	When the non-conservative force is derived from a dissipation function $\mathcal{F}$ where $\bb{F_f} = -\grad_v \mathcal{F}$, the langragian is 
	$$ \frac{d}{dt} \left(\frac{\partial L}{\partial \dot{q}_j}\right) - \frac{\partial L}{\partial q_j} + \frac{\partial \mathcal{F}}{\partial  \dot{q}_j} = 0 $$
\end{theorem}

\section{Variational Principle and Lagrange's Equation}



\begin{theorem}[Frechet Derivative]
	\textnormal{
Consider a functional $J[y] = \int_a^b L(y, y', x) dx$ and its increment while holding $y$ fixed is then a function of $h$:
 $$\Delta J[h] = \delta J[h]  + \epsilon||h||$$
It is the sum of a linear functional (\textbf{variation}) of $J$ and a higher-than-first-order term $\epsilon||h||$ \\
When $||h||$ is small and goes to 0, $\epsilon$ also goes to 0. Then we say the functional is \textbf{differentiable}.
}
\end{theorem}
However, this notion of functional differential is so strong it may not exist and in those cases a weaker notion, like the Gateaux derivative is preferred. In many practical cases, the functional differential is defined as the \textbf{directional derivative}

\begin{theorem}[Gateaux derivative]
  A weaker version of Frechet Derivative. $$
    \delta J[y, h] = \lim_{\alpha \rightarrow 0} \frac{J[y + \alpha h] - J[y]}{\alpha} = \left[\frac{d J[y + \alpha h]}{d \alpha} \right]_{\alpha =0}
  $$
\end{theorem}

To derive the lagrange's equation, suppose $y_*$ is a stationary value for $J[y] = \int^a_b L(y, y', x)dx$. Then we consider $y(x; \alpha) = y_* + \alpha h(x)$ where $h(x)$ is any function with $f(a) = f(b) = 0$.
$$ 
 J(\alpha) = \int^b_a L(y_* + \alpha h, y_*' + \alpha h', x)dx \\
$$
Then $J$ is a function of $\alpha$, and 
$$ \delta J[y, h] \equiv \left(\frac{dJ}{d\alpha}\right)_{\alpha = 0} = 0 $$
because $y(x;\alpha = 0) = y_*$ a stationary value. So the functional's change with $\alpha$ should be 0.  So we arrive at the Gateaux derivative.
Then we just calculate the derivative and integrate by parts:
$$
\begin{aligned}
	 \delta J[y, h] &= \frac{dJ}{d\alpha} \atval_{\alpha = 0} \\
	   &= \int^b_a \frac{\partial L}{\partial y}\frac{\partial y}{\partial \alpha} + \frac{\partial L}{\partial y'}\frac{\partial y'}{\partial \alpha} dx \\
	   &= \int^b_a \frac{\partial L}{\partial y}h + \frac{\partial L}{\partial y'}h' dx \\
	   &=  \int^b_a \frac{\partial L}{\partial y}h dx + \int^b_a \frac{\partial L}{\partial y'}h' dx \\
	   &=  \int^b_a \frac{\partial L}{\partial y}h dx + \left(h\frac{\partial L}{\partial y'}\right)^b_a - \int^b_a \frac{d}{dx} \frac{\partial L}{\partial y'}h dx \\
	   &= \int^b_a \left(\frac{\partial L}{\partial y} - \frac{d}{dx} \frac{\partial L}{\partial y'}\right) h dx, \quad \left(h(a)=h(b)=0 \Rightarrow \left(h\frac{\partial L}{\partial y'}\right)^b_a = 0\right)\\
	   &= 0
\end{aligned}
$$
Since $h$ is arbitrary, if the other term has a non zero region, $h$ can be defined as a bump in that region and 0 everywhere else. So we have lagrange's equation.
$$\frac{\partial L}{\partial y} - \frac{d}{dx} \frac{\partial L}{\partial y'} = 0$$
For the general case where $J = \int_a^b L(q_1,\ldots, q_n, \dot{q_1}, \ldots \dot{q}_n, t)dt$, we can just take $q_i(t, \alpha) = q_{i,*} + \alpha h_i$ and equates $\delta J = 0$ and apply the same procedure as above.
\begin{theorem}[Hamilton's Principle]
	\textnormal{The motion of a system in configuration space is such that  $$ 
		\delta I = \delta \int_{t_1}^{t_2} L(q_1, \ldots, q_n, \dot{q}_1, \ldots, \dot{q}_n, t) dt = 0
		$$ 
		Note that when the constraint is holonomic,  Hamilton's principle is necessary and sufficient condition for Lagrange's equation
		$$ \frac{\partial L}{\partial q_i} -  \frac{d}{dt}\left(\frac{\partial L}{\partial \dot{q}_i}\right) = 0, \quad \textnormal{for $i$ from $1$ to $n$}$$
	}
\end{theorem}

\subsection{Hamilton's Principle with Constraints}
To extend to certain nonholonomic constraints,  for $m$ constraints $f_\alpha$. We find the stationary point for the modified version of the functional: 
$$ \begin{aligned} 
	&\quad I = \int^b_a L + \sum_{\alpha = 1}^m \lambda_\alpha f_\alpha dt  \\
	&\Rightarrow  \delta I[y, h]  \\
	&= \frac{d}{d\epsilon} \int^b_a L + \sum_{\alpha = 1}^m \lambda_\alpha f_\alpha dt \\
	&=  \frac{d}{d\epsilon} \left(\int^b_a L(q_i + \epsilon h_i, \dot{q}_i + \epsilon \dot{h}_i, t) dt  \right) +   \frac{d}{d\epsilon}  \left(\int_a^b \sum_{\alpha = 1}^m \lambda_\alpha f_\alpha (q_i + \epsilon h_i)dt \right) \\
	&= \int_a^b \sum_i \left( \frac{\partial L}{\partial q_i} -  \frac{d}{dt}\left(\frac{\partial L}{\partial \dot{q}_i}\right) \right) h_i dt + \int_a^b  \sum_{\alpha = 1}^m \sum_i \lambda_\alpha \frac{\partial f_\alpha}{\partial q_i}h_i dt  \\
	&= \int_a^b \sum_i \left( \frac{\partial L}{\partial q_i} -  \frac{d}{dt}\left(\frac{\partial L}{\partial \dot{q}_i}\right) +  \sum_{\alpha = 1}^m \lambda_\alpha \frac{\partial f_\alpha}{\partial q_i} \right) h_i dt \\
	&= 0
\end{aligned}$$ 

\begin{theorem}
\textnormal{
 $$  \frac{\partial L}{\partial q_i} -  \frac{d}{dt}\left(\frac{\partial L}{\partial \dot{q}_i}\right) +  \sum_{\alpha = 1}^m \lambda_\alpha \frac{\partial f_\alpha}{\partial q_i} = 0 $$
Note that for applied forces that don't stem from a potential, we have a $Q_i$ term which in this case is the lagrange multipler term so:
$$ \frac{\partial L}{\partial q_i} -  \frac{d}{dt}\left(\frac{\partial L}{\partial \dot{q}_i}\right) = Q_i = - \sum_{\alpha = 1}^m \lambda_\alpha \frac{\partial f_\alpha}{\partial q_i} $$ 
}
\end{theorem}
